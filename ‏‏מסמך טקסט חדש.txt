    public static void assignFlight(Flight[] flights, Passenger p) {
        int x = p.getKm();
        int i = 0;
        int po = 0;
        int o = 0;
        int l = Integer.MAX_VALUE;
        for (Flight flight : flights) {
            o = flight.getKm() + x;
            if (Math.min(o, l) == o && o > 20000) {
                l = o;
                po = i;
            }
            i++;
        }
        p.setKm(o);
        if (!p.getPermPass())
            p.setPermPass(true);
        if (flights[po].getCapacity() < flights[po].getPassengersNum()) {
            Node<Passenger> t = flights[po].getPassengers();
            t.setNext(new Node<Passenger>(p));
            flights[po].setPassengers(t);
            return;
        }
        Queue<Passenger> u = flights[po].getWaiting();
        u.insert(p);
        flights[po].setWaiting(u);
    }

    public static void cancelAndReplace(Flight[] flights, int i, Passenger canceling) {
        flights[i].setPassengers(del(flights[i].getPassengers(), canceling));
        canceling.setKm(flights[i].getKm() - canceling.getKm());
        Queue<Passenger> waitingQueue = flights[i].getWaiting();
        Passenger replacement = null;
        int maxKm = -1;
        Passenger frequentFlyer = null;
        Queue<Passenger> tempQueue = new Queue<>();

        while (!waitingQueue.isEmpty()) {
            Passenger p = waitingQueue.remove();
            if (p.getPermPass() && p.getKm() > maxKm) {
                maxKm = p.getKm();
                frequentFlyer = p;
            }
            tempQueue.insert(p);
        }
        if (frequentFlyer != null) {
            replacement = frequentFlyer;
            // rebuild waitingQueue from tempQueue, skipping the first occurrence of
            // frequentFlyer
            Queue<Passenger> newWaiting = new Queue<>();
            boolean skipped = false;
            while (!tempQueue.isEmpty()) {
                Passenger p = tempQueue.remove();
                if (!skipped && p == frequentFlyer) {
                    skipped = true;
                    continue;
                }
                newWaiting.insert(p);
            }
            waitingQueue = newWaiting;
        } else {

            replacement = tempQueue.remove();
            // remaining elements in tempQueue become the waiting queue
            waitingQueue = tempQueue;
        }

        // 4. הוספת הנוסע שהוחלף לרשימת הנוסעים
        flights[i].setPassengers(addToPassengers(flights[i].getPassengers(), replacement));
        flights[i].setWaiting(waitingQueue);
    }

    public static <T> Node<T> del(Node<T> p, T x) { // ex6
        Node<T> head = new Node<>(null, p);
        p = head;
        while (p.hasNext()) {
            if (p.getNext().getValue() == x) {
                p.setNext(p.getNext().getNext());
                break;
            } else
                p = p.getNext();
        }
        return head.getNext();
    }

    public static <T> Queue<T> ex_1(Queue<T> q) {
        Queue<T> co = new Queue<>();
        Queue<T> te = new Queue<>();
        while (!q.isEmpty()) {
            T vl = q.remove();
            co.insert(vl);
            te.insert(vl);
        }
        while (!te.isEmpty())
            q.insert(te.remove());
        return co;
    }

    public static Node<Passenger> addToPassengers(Node<Passenger> passengers, Passenger newPassenger) {
        if (passengers == null) {
            return new Node<Passenger>(newPassenger);
        }
        Node<Passenger> current = passengers;
        while (current.getNext() != null) {
            current = current.getNext();
        }
        current.setNext(new Node<Passenger>(newPassenger));

        return passengers;
    }
